\subsection*{Commands for Latin Letters (A-Z, a-z)}

\begin{itemize}
    \item \texttt{\textbackslash{}mathit} — italic ($\mathit{A} \ldots \mathit{Z}$ $\mathit{a} \ldots \mathit{z}$)
    \item \texttt{\textbackslash{}mathbf} — bold upright ($\mathbf{A} \ldots \mathbf{Z}$ $\mathbf{a} \ldots \mathbf{z}$)
    \item \texttt{\textbackslash{}mathsf} — sans-serif ($\mathsf{A} \ldots \mathsf{Z}$ $\mathsf{a} \ldots \mathsf{z}$)
    \item \texttt{\textbackslash{}mathtt} — typewriter ($\mathtt{A} \ldots \mathtt{Z}$ $\mathtt{a} \ldots \mathtt{z}$)
    \item \texttt{\textbackslash{}mathbfit} — bold italic ($\mathbfit{A} \ldots \mathbfit{Z}$ $\mathbfit{a} \ldots \mathbfit{z}$)
    \item \texttt{\textbackslash{}mathcal} — calligraphic ($\mathcal{A} \ldots \mathcal{Z}$)
    \item \texttt{\textbackslash{}mathscr} — script ($\mathscr{A} \ldots \mathscr{Z}$)
    \item \texttt{\textbackslash{}mathbb} — blackboard bold ($\mathbb{A} \ldots \mathbb{Z}$)
    \item \texttt{\textbackslash{}mathfrak} — fraktur ($\mathfrak{A} \ldots \mathfrak{Z}$ $\mathfrak{a} \ldots \mathfrak{z}$)
\end{itemize}
