% 1. label_definition - defining a label
\section{Introduction} \label{sec:intro} % Label definition for a section

\section{Conclusion}
\label{sec:conclusion}                % Label definition for another section

\begin{equation}
a = b
\label{eq:first}                      % Label definition for first equation
\end{equation}

\begin{equation}
c = d
\label{eq:second}                     % Label definition for second equation
\end{equation}

\begin{figure}
\centering
\caption[First image]{First figure}
\label{fig:start}                     % Label definition for first figure
\end{figure}

\begin{figure}
\centering

\label{fig:end}                       % Label definition for last figure
\end{figure}

% 3. label_reference_range - range of references
Sections \crefrange{sec:intro}{sec:conclusion}
Equations \crefrange{eq:first}{eq:second}
Figures \crefrange{fig:start}{fig:end}

\Crefrange{sec:intro}{sec:conclusion}            % Capitalized at start of sentence
\cpagerefrange{sec:intro}{sec:conclusion}        % Page number range
\Cpagerefrange{sec:intro}{sec:conclusion}        % Capitalized page range

% Starred versions
\crefrange*{sec:intro}{sec:conclusion}           % Without hyperlinks
\Crefrange*{sec:intro}{sec:conclusion}           % Starred capitalized
