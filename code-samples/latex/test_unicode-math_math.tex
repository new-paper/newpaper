\subsection*{Commands for Latin Letters (A-Z, a-z)}

\begin{itemize}
    \item \texttt{\textbackslash{}mathup} — upright ($\mathup{A} \ldots \mathup{Z}$ $\mathup{a} \ldots \mathup{z}$)
    \item \texttt{\textbackslash{}mathit} — italic ($\mathit{A} \ldots \mathit{Z}$ $\mathit{a} \ldots \mathit{z}$)
    \item \texttt{\textbackslash{}mathbf} — bold upright ($\mathbf{A} \ldots \mathbf{Z}$ $\mathbf{a} \ldots \mathbf{z}$)
    \item \texttt{\textbackslash{}mathsf} — sans-serif ($\mathsf{A} \ldots \mathsf{Z}$ $\mathsf{a} \ldots \mathsf{z}$)
    \item \texttt{\textbackslash{}mathtt} — typewriter ($\mathtt{A} \ldots \mathtt{Z}$ $\mathtt{a} \ldots \mathtt{z}$)
    \item \texttt{\textbackslash{}mathbfup} — bold upright ($\mathbfup{A} \ldots \mathbfup{Z}$ $\mathbfup{a} \ldots \mathbfup{z}$)
    \item \texttt{\textbackslash{}mathbfit} — bold italic ($\mathbfit{A} \ldots \mathbfit{Z}$ $\mathbfit{a} \ldots \mathbfit{z}$)
    \item \texttt{\textbackslash{}mathsfup} — sans-serif ($\mathsfup{A} \ldots \mathsfup{Z}$ $\mathsfup{a} \ldots \mathsfup{z}$)
    \item \texttt{\textbackslash{}mathsfit} — sans-serif italic ($\mathsfit{A} \ldots \mathsfit{Z}$ $\mathsfit{a} \ldots \mathsfit{z}$)
    \item \texttt{\textbackslash{}mathbfsf} — sans-serif bold ($\mathbfsf{A} \ldots \mathbfsf{Z}$ $\mathbfsf{a} \ldots \mathbfsf{z}$)
    \item \texttt{\textbackslash{}mathbfsfup} — sans-serif bold ($\mathbfsfup{A} \ldots \mathbfsfup{Z}$ $\mathbfsfup{a} \ldots \mathbfsfup{z}$)
    \item \texttt{\textbackslash{}mathbfsfit} — sans-serif bold italic ($\mathbfsfit{A} \ldots \mathbfsfit{Z}$ $\mathbfsfit{a} \ldots \mathbfsfit{z}$)
    \item \texttt{\textbackslash{}mathcal} — calligraphic ($\mathcal{A} \ldots \mathcal{Z}$ $\mathcal{a} \ldots \mathcal{z}$)
    \item \texttt{\textbackslash{}mathbfcal} — bold calligraphic ($\mathbfcal{A} \ldots \mathbfcal{Z}$ $\mathbfcal{a} \ldots \mathbfcal{z}$)
    \item \texttt{\textbackslash{}mathscr} — script ($\mathscr{A} \ldots \mathscr{Z}$ $\mathscr{a} \ldots \mathscr{z}$)
    \item \texttt{\textbackslash{}mathbfscr} — bold script ($\mathbfscr{A} \ldots \mathbfscr{Z}$ $\mathbfscr{a} \ldots \mathbfscr{z}$)
    \item \texttt{\textbackslash{}mathbb} — blackboard bold ($\mathbb{A} \ldots \mathbb{Z}$ $\mathbb{a} \ldots \mathbb{z}$)
    \item \texttt{\textbackslash{}mathbbit} — blackboard bold ($\mathbbit{A} \ldots \mathbbit{Z}$ $\mathbbit{a} \ldots \mathbbit{z}$)
    \item \texttt{\textbackslash{}mathfrak} — fraktur ($\mathfrak{A} \ldots \mathfrak{Z}$ $\mathfrak{a} \ldots \mathfrak{z}$)
    \item \texttt{\textbackslash{}mathbffrak} — bold fraktur ($\mathbffrak{A} \ldots \mathbffrak{Z}$ $\mathbffrak{a} \ldots \mathbffrak{z}$)
\end{itemize}
