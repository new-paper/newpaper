\subsection*{Usage example in formulas}
\[
\mathbf{A} + \mathbf{B} = \mathbf{C}
\]
\[
\mathbf{F} = m \mathbf{a}
\]
\[
\mathbfit{A} + \mathbfit{B} = \mathbfit{C}
\]
\[
\mathbfit{F} = m \mathbfit{a}
\]
\[
\bm{A} + \bm{B} = \bm{C}
\]
\[
\bm{F} = m \bm{a}
\]
\[
\bm{\alpha} + \bm{\beta} = \bm{\Gamma} \cdot \bm{\Omega} = \bm{\Pi} \cdot \bm{\Sigma}
\]
\[
\bm{\upalpha} + \bm{\upbeta} = \bm{\Upgamma} \cdot \bm{\Upomega} = \bm{\Uppi} \cdot \bm{\Upsigma}
\]
